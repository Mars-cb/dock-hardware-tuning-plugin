cmake_minimum_required(VERSION 3.13)
project(dock-hardware-tuning-plugin VERSION 0.2.9 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

find_package(Qt5 5.11 REQUIRED COMPONENTS Widgets Svg)
find_package(DtkWidget 5.6 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(UDEV REQUIRED IMPORTED_TARGET libudev)
find_path(DDE_DOCK_INCLUDE_DIR pluginsiteminterface_v2.h PATH_SUFFIXES dde-dock)
if(NOT DDE_DOCK_INCLUDE_DIR)
    set(DDE_DOCK_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/dde-dock-5.9")
endif()

add_library(dock-hardware-tuning-plugin MODULE plugin.cpp resources.qrc)
target_include_directories(dock-hardware-tuning-plugin PRIVATE "${DDE_DOCK_INCLUDE_DIR}")
target_link_libraries(dock-hardware-tuning-plugin PRIVATE Qt5::Widgets Qt5::Svg Dtk::Widget PkgConfig::UDEV)
target_compile_options(dock-hardware-tuning-plugin PRIVATE -Wall -Wextra)

add_executable(dock-hardware-tuning-plugin-preview preview.cpp)
target_include_directories(dock-hardware-tuning-plugin-preview PRIVATE "${DDE_DOCK_INCLUDE_DIR}")
target_link_libraries(dock-hardware-tuning-plugin-preview PRIVATE Qt5::Widgets)
target_compile_options(dock-hardware-tuning-plugin-preview PRIVATE -Wall -Wextra)

install(TARGETS dock-hardware-tuning-plugin LIBRARY DESTINATION /usr/lib/dde-dock/plugins)
install(PROGRAMS controller.py DESTINATION /usr/libexec RENAME dock-hardware-tuning-plugin-helper)
install(FILES packaging/io.github.DockHardwareTuning.conf DESTINATION /usr/share/dbus-1/system.d)
install(FILES packaging/io.github.DockHardwareTuning.service DESTINATION /usr/share/dbus-1/system-services)
install(FILES packaging/dock-hardware-tuning.service DESTINATION /lib/systemd/system)
